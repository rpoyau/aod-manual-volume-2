\section{Seat Compatibility, Enclosures, and Coord-0 Grouping}\label{sec:coord0-enclosures}

\subsection{Even enclosure sequence and cumulative coord-0 boundaries}\label{subsec:coord0-grouping-derivation}

\paragraph{Setup.}
Import the seat-compatibility and shell/enclosure routing hierarchy of \mainref{\S5.7.1--\S5.7.13} and \mainref{\S5.8}:
\begin{enumerate}[nosep]
  \item \textbf{Current-side double forms (\mainref{\S5.7.2--\S5.7.3}):} $E_2 = \mathrm{duon}$, $E_2^0 = \text{open-seat duon}$, and $E_2^\pm = \mathrm{duad}$ (the seat-taken state of $E_2$).
  \item \textbf{Default inner enclosure (\mainref{\S5.7.4}):} $E_4 = \mathrm{tetron}$, the default four-seat inner enclosure anchored at local origin $0\vdash$.
  \item \textbf{Outer even support enclosures (\mainref{\S5.7.5--\S5.7.9}, \mainref{App.~B.3}):} $\mathsf{C}_{2m}$ denotes the outer support enclosure of even order $2m$ for $m \ge 3$, beginning with $\mathsf{C}_6 = \mathrm{hexon}$ ($\mathsf{H}_3 + \mathsf{H}_3 = 6$), $\mathsf{C}_8 = \mathrm{octon}$, $\mathsf{C}_{10} = \mathrm{decon}$, and continuing through higher even orders $\mathsf{C}_{2m}$ ($m > 5$).
\end{enumerate}

\paragraph{Derivation.}
At coordinate $0$, a coord-0 shell-stack declaration has the canonical form (\mainref{\S5.7.13})
\begin{equation}
0\vdash :: Z : A : 0 : L\langle a : b : c :: L\rangle,
\label{eq:matter1-coord0-declaration}
\end{equation}
and a filled shared support enclosure of order $2m$ ($m \ge 3$) has the canonical form (\mainref{\S5.7.10, \S5.7.13, App.~B.4.3})
\begin{equation}
0\vdash :: Z : A : 0 : 2m\langle m : m : 0 :: 2m\rangle \;=\; \mathrm{Tritrio}^m :: \mathrm{Tetratrio}^m :: \operatorname{out}(2m).
\label{eq:matter1-filled-shared-enclosure}
\end{equation}
Summing the even seat capacities $2k$ from the first outer enclosure $k = 3$ ($\mathsf{C}_6$) through $k = m$ yields the cumulative coord-0 group boundary (\mainref{\S5.7.13}):
\begin{equation}
\Gamma_m = \sum_{k=3}^{m} 2k = m(m+1) - 6, \qquad \Gamma_2 = 0.
\label{eq:matter1-gamma-m}
\end{equation}
The strict interval condition
\begin{equation}
\Gamma_{m-1} < Z \le \Gamma_m
\label{eq:matter1-group-selection}
\end{equation}
uniquely selects the coord-0 support-enclosure group $\mathsf{C}_{2m}$ and local seat index $s = Z - \Gamma_{m-1} \in \{1, \dots, 2m\}$ for every integer $Z \ge 1$. Along each ring $\mathsf{C}_{2m}$, the $2m$ seats partition into:
\begin{equation}
a(s) = \left\lceil \frac{s}{2}\right\rceil, \qquad b(s) = \left\lfloor \frac{s}{2}\right\rfloor, \qquad c(s) = s \bmod 2,
\label{eq:matter1-seat-abc}
\end{equation}
so that $s = 2$ on $\mathsf{C}_6$ gives $1{:}1{:}0\,{:}{:}\,6$ (\mainref{\S5.8.1}), odd seats $s = 2j-1$ carry one open-seat duon ($c = 1, E_2^0$), even seats $s = 2j$ close $j$ seated duads ($a = b = j, c = 0, E_2^\pm$), half-shell turnaround occurs at $s = m$ on the 6~o'clock $H_\mu$ hinge axis ($\Pi_\alpha \to \Pi_\Omega$), and $s = 2m$ ($Z = \Gamma_m$) locks the filled shared enclosure $m{:}m{:}0\,{:}{:}\,2m$ on the 12~o'clock spine.

\paragraph{Specialization.}
Evaluating \eqnref{eq:matter1-gamma-m}--\eqnref{eq:matter1-seat-abc} across $m = 3 \dots 11$ yields the nine concentric outer support enclosures $\mathsf{C}_6 \dots \mathsf{C}_{22}$ summarized in \tabref{tab:matter1-enclosure-groups} and rendered in \figref{fig:matter1-concentric-wheel}. Across $m = 3 \dots 11$, total coord-0 capacity is $\Gamma_{11} = 11(12) - 6 = 126$ slots; the admitted elemental ladder $Z = 1 \dots 118$ occupies the exact rational fraction $118/126 = 59/63$ of $\Gamma_{11}$ (and $14/22 = 7/11$ on $\mathsf{C}_{22}$), while slots $Z = 119 \dots 126$ on $\mathsf{C}_{22}$ form the unretained open cutoff ($\mathrm{TTL}_{\mathcal{C}}(\partial F,\omega) < |\omega|$).

\paragraph{Calibration.}
By \mainref{\S5.2.4, (90)} and \mainref{\S5.17.4}, field-count scaling $\mathcal{O}_{\partial F}(\mathcal{B}) = n_F(\mathcal{B})\log n_F(\mathcal{B})$ (for $n_F(\mathcal{B}) \ge 1$) and seat-compatible shell/enclosure routing (\eqnref{eq:matter1-coord0-declaration}--\eqnref{eq:matter1-seat-abc}) are distinct calculations on the same declared boundary/window:
\begin{equation}
\begin{aligned}
B &\to \mathcal{F}_B \to n_F(B) \to \ADAR_{i\leftarrow j} \to \mathcal{O}_{\partial F}(B) \to \SADAR,\\
\mathsf{C}_{2m} &\to X_{\mathrm{shedding}} \to \operatorname{SheddicPath}(X_{\mathrm{shedding}}).
\end{aligned}
\label{eq:matter1-calculation-split}
\end{equation}

\paragraph{Falsification.}
Assigning an element $Z$ to an odd support enclosure, shifting $\Gamma_m$ away from $m(m+1)-6$, or conflating $n_F \log n_F$ field-count weighting with $\mathsf{C}_{2m}$ seat routing fails the seat/enclosure routing audit (\mainref{\S5.17.4}).
